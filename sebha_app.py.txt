import streamlit as st
import json
import os

SAVE_FILE = "sebha_web_counts.json"

# تحميل البيانات إن وجدت
def load_counts():
    if os.path.exists(SAVE_FILE):
        with open(SAVE_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    return {"سبحان الله": 0, "الحمد لله": 0, "الله أكبر": 0}

# حفظ البيانات
def save_counts(data):
    with open(SAVE_FILE, "w", encoding="utf-8") as f:
        json.dump(data, f)

# -----------------------
st.set_page_config(page_title="مسبحة إلكترونية", page_icon="🕌", layout="centered")
st.title("🌙 مسبحة إلكترونية")

counts = load_counts()

# عرض العداد والأزرار لكل نوع
for tasbeeh in ["سبحان الله", "الحمد لله", "الله أكبر"]:
    st.subheader(tasbeeh)
    cols = st.columns([2, 1])
    with cols[0]:
        st.markdown(f"<h1 style='text-align: center;'>{counts[tasbeeh]}</h1>", unsafe_allow_html=True)
    with cols[1]:
        if st.button(f"تسبيح {tasbeeh}", key=tasbeeh):
            counts[tasbeeh] += 1
            save_counts(counts)
            st.experimental_rerun()

st.divider()

# زر إعادة الضبط
if st.button("🔄 إعادة الضبط"):
    for key in counts:
        counts[key] = 0
    save_counts(counts)
    st.experimental_rerun()
